# 1. 装 Wrangler (Install Wrangler)
npm install -g wrangler

# 2. 登录 Cloudflare (Log in to Cloudflare)
wrangler login

#3.进入项目路径 (Enter the project path)
cd 你的项目绝对路径 (your project absolute path)

#4.正式部署项目 (Officially deploy the project)
wrangler pages deploy public --project-name=你的项目名字 (your project name)
